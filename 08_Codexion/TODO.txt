### 2. Implementação do `init.c` (ou no `utils.c`)

Crie a lógica para alocação, amarração circular dos dongles e limpeza de memória:

```c
#include "coders.h"

static void init_dongles(t_workspace *ws)
{
    long    i;

    ws->dongles = malloc(sizeof(t_dongle) * ws->number_of_coders);
    if (!ws->dongles)
        error_exit("Memory allocation failed for dongles!");
    
    i = 0;
    while (i < ws->number_of_coders)
    {
        ws->dongles[i].dongle_id = i + 1;
        if (pthread_mutex_init(&ws->dongles[i].dongle, NULL) != 0)
            error_exit("Failed to initialize dongle mutex!");
        i++;
    }
}

static void init_coders(t_workspace *ws)
{
    long    i;

    ws->coders = malloc(sizeof(t_coder) * ws->number_of_coders);
    if (!ws->coders)
        error_exit("Memory allocation failed for coders!");

    i = 0;
    while (i < ws->number_of_coders)
    {
        ws->coders[i].id = i + 1;
        ws->coders[i].number_compiles = 0;
        ws->coders[i].bournout = false;
        ws->coders[i].last_compile_time = 0;
        ws->coders[i].workspace = ws;

        // Distribuição circular dos dongles na mesa:
        // Coder i pega o dongle i na esquerda e o dongle (i + 1) % total na direita
        ws->coders[i].left_dongle = &ws->dongles[i];
        ws->coders[i].right_dongle = &ws->dongles[(i + 1) % ws->number_of_coders];

        i++;
    }
}

void init_workspace(t_workspace *ws)
{
    ws->start_simulation = 0;
    ws->end_simulation = false;

    if (pthread_mutex_init(&ws->log_mutex, NULL) != 0)
        error_exit("Failed to initialize log mutex!");

    init_dongles(ws);
    init_coders(ws);
}

// Função para destruir mutexes e liberar memória ao final da execução
void cleanup_workspace(t_workspace *ws)
{
    long    i;

    if (ws->dongles)
    {
        i = 0;
        while (i < ws->number_of_coders)
        {
            pthread_mutex_destroy(&ws->dongles[i].dongle);
            i++;
        }
        free(ws->dongles);
    }
    pthread_mutex_destroy(&ws->log_mutex);
    if (ws->coders)
        free(ws->coders);
}

```
